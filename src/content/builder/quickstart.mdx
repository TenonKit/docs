export const meta = {
  group: "Get started",
  title: "Quickstart",
  description: "Build your first page, change its look and export it.",
};

<Steps>
  <Step n={1} title="Create a project">
    On Home, choose **New project**. Keep **Website** and **Start with an example page** ticked, give it a name and create it.
  </Step>
  <Step n={2} title="Add components">
    Drag a component from the left onto the page, or click it to add it below the selected one. The panel on the right shows whatever you’ve selected.
  </Step>
  <Step n={3} title="Change the look">
    Click a heading inside a component and open the **Design** tab. Change its colour or size and watch the canvas update.
  </Step>
  <Step n={4} title="Preview and export">
    Choose **Preview** to open the page in your browser, then **Export** to save it as HTML or a React project.
  </Step>
</Steps>

<Callout type="tip">
  There’s no save button to remember. Every change is saved a moment after you make it, and **Saved** next to the project name confirms it.
</Callout>
